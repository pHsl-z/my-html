function exportToPdf() { const content = htmlEditor.value.trim(); if (!content) { showToast('请先输入内容', 'error'); return; } try { const previewDoc = previewFrame.contentDocument || previewFrame.contentWindow.document; if (!previewDoc || !previewDoc.body || !previewDoc.body.innerHTML.trim()) { showToast('预览内容为空，请先输入内容', 'error'); return; } if (typeof window.jspdf === 'undefined' || typeof html2canvas === 'undefined') { showToast('PDF库加载中，请稍后重试', 'error'); return; } showToast('正在生成PDF...', 'info'); const container = document.createElement('div'); container.style.cssText = 'position:fixed;left:0;top:0;width:794px;background:#fff;padding:40px;font-family:-apple-system,BlinkMacSystemFont,"Segoe UI",Roboto,Arial,sans-serif;color:#000;line-height:1.6;z-index:-1;'; container.innerHTML = previewDoc.body.innerHTML; document.body.appendChild(container); const title = (content.match(/]*>([\s\S]*?)<\/title>/i) || [])[1]?.trim() || 'export'; const filename = title.replace(/[^a-zA-Z0-9\u4e00-\u9fa5_-]/g, '_') + '.pdf'; html2canvas(container, { scale: 2, useCORS: true, logging: false, backgroundColor: '#ffffff', width: 794, windowWidth: 794 }).then(canvas => { const { jsPDF } = window.jspdf; const imgWidth = 210; const imgHeight = (canvas.height * imgWidth) / canvas.width; const pageHeight = 297; const pdf = new jsPDF('p', 'mm', 'a4'); let position = 0; let remainingHeight = imgHeight; while (remainingHeight > 0) { const sourceY = position * (canvas.width / imgWidth); const sourceHeight = pageHeight * (canvas.width / imgWidth); const pageCanvas = document.createElement('canvas'); pageCanvas.width = canvas.width; pageCanvas.height = Math.min(sourceHeight, canvas.height - sourceY); const ctx = pageCanvas.getContext('2d'); ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, pageCanvas.width, pageCanvas.height); ctx.drawImage(canvas, 0, sourceY, pageCanvas.width, pageCanvas.height, 0, 0, pageCanvas.width, pageCanvas.height); const pageImgData = pageCanvas.toDataURL('image/jpeg', 0.95); if (position > 0) pdf.addPage(); pdf.addImage(pageImgData, 'JPEG', 0, 0, imgWidth, (pageCanvas.height * imgWidth) / canvas.width); remainingHeight -= pageHeight; position += pageHeight; } const pdfBlob = pdf.output('blob'); if (navigator.share && navigator.canShare && navigator.canShare({ files: [new File([pdfBlob], filename, { type: 'application/pdf' })] })) { const file = new File([pdfBlob], filename, { type: 'application/pdf' }); navigator.share({ files: [file], title: title }).then(() => { showToast('PDF导出成功', 'success'); }).catch(err => { if (err.name !== 'AbortError') { pdf.save(filename); } }); } else { pdf.save(filename); showToast('PDF导出成功', 'success'); } try { document.body.removeChild(container); } catch(e) {} }).catch(err => { showToast('PDF生成失败', 'error'); try { document.body.removeChild(container); } catch(e) {} }); } catch (error) { showToast('导出失败，请重试', 'error'); } }
